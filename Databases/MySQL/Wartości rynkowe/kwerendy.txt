create procedure zmiana(in id_zaw int,in nowa_wartość numeric(9,3), out p_ostatnia_zmiana numeric(9,3), out p_wzrost_spadek char(2), out id_zaw_out int)
begin 
DECLARE p_wartość_rynkowa numeric(9,3) DEFAULT 0.0;
set id_zaw_out = id_zaw;
set p_wzrost_spadek= 0;
SELECT wartość_rynkowa into p_wartość_rynkowa from polska where id_zawodnika = id_zaw;
if nowa_wartość > p_wartość_rynkowa THEN
update polska set wartość_rynkowa=nowa_wartość where id_zawodnika=id_zaw;
set p_ostatnia_zmiana = nowa_wartość - p_wartość_rynkowa;
set p_wzrost_spadek="+";
ELSE
update polska set wartość_rynkowa=nowa_wartość where id_zawodnika=id_zaw;
set p_ostatnia_zmiana = p_wartość_rynkowa - nowa_wartość ;
set p_wzrost_spadek="-";
end if;
end$$
